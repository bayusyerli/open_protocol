    ?pathogen wdt:P2975 ?host .
    ?host wdt:P225 ?hostName .
    FILTER(?hostName IN ("Capsicum annuum","Capsicum frutescens","Solanum lycopersicum",
                         "Solanum tuberosum","Allium cepa","Allium ascalonicum",

  } ORDER BY ?hostName ?pathogenLabel
